#---出力画像を 500×500 に固定
#---CPUスレッド数を制限してPC負荷を軽減
#---Cyclesのサンプル数を減らして高速化
#---処理するArea範囲を指定できるようにした
#---すでに出力済みの画像はスキップ
#---90°・270°の両方向を必ず出力
#---エラーが出ても次のAreaへ進むようにした
#---ファイル名をWindowsで安全な形式に修正
#--一言でいうと、元の撮影ロジックは維持しつつ、処理を軽くして途中再開しやすくした形です。

import bpy
import csv
import math
import mathutils
import os
import time


# ============================================================
# 基本設定
# ============================================================

CSV_PATH = r"D:/yasuda_work/merged_data_tb2_修正版_ratio1.3.csv"
OUTPUT_FOLDER = r"D:/yasuda_work/sadoku/fig_ratio1.3"

# 処理対象Area
START_AREA = 1
END_AREA = 2279


# ============================================================
# 負荷軽減設定
# ============================================================

# CPUスレッド数
# UPS負荷が高ければ 8 -> 6 -> 4 と下げてください
CPU_THREADS = 4

# Cyclesサンプル数
CYCLES_SAMPLES = 16

# 出力解像度
RESOLUTION_X = 500
RESOLUTION_Y = 500

# 各レンダリング後の休止秒
RENDER_INTERVAL = 0.5


# ============================================================
# 出力フォルダ作成
# ============================================================

if not os.path.exists(OUTPUT_FOLDER):
    os.makedirs(OUTPUT_FOLDER)


# ============================================================
# カメラ準備
# ============================================================

if not bpy.data.objects.get("Camera"):

    camera_data = bpy.data.cameras.new("Camera")
    camera = bpy.data.objects.new("Camera", camera_data)

    bpy.context.scene.collection.objects.link(camera)
    bpy.context.scene.camera = camera

else:

    camera = bpy.data.objects["Camera"]


# ============================================================
# Scene設定
# ============================================================

scene = bpy.context.scene
scene.camera = camera

# Cycles
scene.render.engine = 'CYCLES'


# ============================================================
# 魚眼カメラ設定
#
# ★ 等距離射影（Equidistant）
# ============================================================

camera.data.type = 'PANO'

if hasattr(camera.data, "panorama_type"):
    camera.data.panorama_type = 'FISHEYE_EQUIDISTANT'

# 魚眼FOVを180度に設定
if hasattr(camera.data, "fisheye_fov"):
    camera.data.fisheye_fov = math.radians(180)

# FOV設定
camera.data.lens_unit = 'FOV'
camera.data.angle = math.radians(180)


# ============================================================
# レンダリング設定
# ============================================================

scene.render.image_settings.file_format = 'PNG'

scene.render.resolution_x = RESOLUTION_X
scene.render.resolution_y = RESOLUTION_Y
scene.render.resolution_percentage = 100


# ============================================================
# Cycles負荷軽減
# ============================================================

scene.cycles.samples = CYCLES_SAMPLES

scene.cycles.max_bounces = 4
scene.cycles.diffuse_bounces = 2
scene.cycles.glossy_bounces = 2
scene.cycles.transmission_bounces = 2

# デノイズ
scene.cycles.use_denoising = True

# CPUレンダリング
scene.cycles.device = 'CPU'

# CPUスレッド数制限
scene.render.threads_mode = 'FIXED'
scene.render.threads = CPU_THREADS


# ============================================================
# レンダリング関数
# ============================================================

def render_image(area, center, direction, angle_label):

    output_path = os.path.join(
        OUTPUT_FOLDER,
        f"Area_{area}_fisheye_equidistant_view_{angle_label}.png"
    )

    # --------------------------------------------------------
    # 既に画像がある場合はスキップ
    # --------------------------------------------------------

    if os.path.exists(output_path):

        print(
            f"[SKIP] Area {area} "
            f"{angle_label}° は既に存在します"
        )

        return

    # --------------------------------------------------------
    # カメラ位置
    # --------------------------------------------------------

    camera.location = center

    # --------------------------------------------------------
    # 元コードと同じカメラ回転処理
    # --------------------------------------------------------

    direction = direction.normalized()

    rot_quat = direction.to_track_quat(
        'Z',
        'Y'
    )

    camera.rotation_euler = rot_quat.to_euler()

    # --------------------------------------------------------
    # 出力先
    # --------------------------------------------------------

    scene.render.filepath = output_path

    # --------------------------------------------------------
    # レンダリング
    # --------------------------------------------------------

    print(
        f"[RENDER] Area {area} "
        f"{angle_label}°"
    )

    bpy.ops.render.render(
        write_still=True
    )

    print(
        f"[SAVE] {output_path}"
    )

    # --------------------------------------------------------
    # PCを少し休ませる
    # --------------------------------------------------------

    if RENDER_INTERVAL > 0:
        time.sleep(RENDER_INTERVAL)


# ============================================================
# CSV読み込み
# ============================================================

processed_count = 0
error_count = 0


with open(
    CSV_PATH,
    newline='',
    encoding='utf-8'
) as csvfile:

    reader = csv.DictReader(csvfile)

    for row in reader:

        try:

            # =================================================
            # Area
            # =================================================

            area = row['Area']
            area_num = int(float(area))

            # 指定範囲外はスキップ
            if (
                area_num < START_AREA
                or area_num > END_AREA
            ):
                continue


            print()
            print("=" * 60)
            print(f"Area {area_num}")
            print("=" * 60)


            # =================================================
            # メッシュ頂点座標
            # =================================================

            vertex_A = mathutils.Vector((
                float(row['xA']),
                float(row['yA']),
                float(row['zA'])
            ))

            vertex_B = mathutils.Vector((
                float(row['xB']),
                float(row['yB']),
                float(row['zB'])
            ))

            vertex_C = mathutils.Vector((
                float(row['xC']),
                float(row['yC']),
                float(row['zC'])
            ))


            # =================================================
            # メッシュ中心
            # =================================================

            x_center = float(row['X_center'])
            y_center = float(row['Y_center'])
            z_center = float(row['Z_center'])

            center = mathutils.Vector((
                x_center,
                y_center,
                z_center
            ))


            # =================================================
            # 法線ベクトル計算
            # 元コードと同じ
            # ============================================================

            edge1 = vertex_B - vertex_A
            edge2 = vertex_C - vertex_A

            normal = edge1.cross(edge2)

            # ゼロ法線対策
            if normal.length < 1e-8:

                print(
                    f"[ERROR] Area {area_num}: "
                    "法線ベクトルが0です"
                )

                error_count += 1
                continue

            normal.normalize()


            # =================================================
            # 元コードの法線反転処理を維持
            # ============================================================

            if normal.dot(center) > 0:
                normal = -normal


            # =================================================
            # 90°方向
            # ============================================================

            look_at_point_90 = center + normal

            direction_90 = (
                look_at_point_90 - center
            ).normalized()

            render_image(
                area_num,
                center,
                direction_90,
                "90"
            )


            # =================================================
            # 270°方向
            # ============================================================

            look_at_point_270 = center - normal

            direction_270 = (
                look_at_point_270 - center
            ).normalized()

            render_image(
                area_num,
                center,
                direction_270,
                "270"
            )


            processed_count += 1


        # =====================================================
        # CSV数値エラー
        # =====================================================

        except ValueError as e:

            error_count += 1

            print(
                f"[ValueError] "
                f"Area {row.get('Area', 'UNKNOWN')}"
            )

            print(
                "Error:",
                e
            )


        # =====================================================
        # その他エラー
        # =====================================================

        except Exception as e:

            error_count += 1

            print(
                f"[Unexpected Error] "
                f"Area {row.get('Area', 'UNKNOWN')}"
            )

            print(
                "Error:",
                e
            )


# ============================================================
# 完了
# ============================================================

print()
print("=" * 60)
print("すべてのメッシュの画像生成が完了しました")
print("処理Area数:", processed_count)
print("エラー数:", error_count)
print("=" * 60)
