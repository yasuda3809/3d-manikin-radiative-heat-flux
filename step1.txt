#--- Step1
#--- csvファイルからblender上にobjを自動作成
#--- (csvファイルの座標とblender上の位置を合わせるため)
#--- エラーが出るが無視する。blender上に作成されたobjを残しておいて、Step2に進む
#--- Blenderでinitialで表示されるBOXのイラストは削除し、メニューバーから「Scripting」を探して、新しいコードからStep1のコードを張り付ける
#--- Runを実行でマネキンモデルの作成が完了

import bpy
import csv
import os

csv_path = "D:/95.python/py360convert-master/merged_data_tb2.csv"

output_obj_path = "D:/95.python/py360convert-master/output_model.obj"

mesh_base_name = "Mesh"
object_base_name = "Object"

bpy.ops.object.select_all(action='SELECT')
bpy.ops.object.delete()

with open(csv_path, newline='', encoding='utf-8') as csvfile:
    reader = csv.DictReader(csvfile)
    for row in reader:
        try:
            vertex_A = (float(row['xA']), float(row['yA']), float(row['zA']))
            vertex_B = (float(row['xB']), float(row['yB']), float(row['zB']))
            vertex_C = (float(row['xC']), float(row['yC']), float(row['zC']))

            vertices = [vertex_A, vertex_B, vertex_C]
            faces = [(0, 1, 2)]

            mesh_name = f"{mesh_base_name}_{row['Area']}"
            object_name = f"{object_base_name}_{row['Area']}"
            mesh = bpy.data.meshes.new(mesh_name)
            obj = bpy.data.objects.new(object_name, mesh)

            mesh.from_pydata(vertices, [], faces)
            mesh.update()

            bpy.context.scene.collection.objects.link(obj)

        except ValueError as e:
            print(f"Error parsing row: {row}, Error: {e}")

bpy.ops.export_scene.obj(filepath=output_obj_path, use_selection=False)

print(f"OBJファイルが作成されました: {output_obj_path}")
